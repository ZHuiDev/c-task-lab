cmake_minimum_required(VERSION 3.16)
project(c_task_lab VERSION 1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/W4 /utf-8)
else()
    add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
add_library(task_core src/task.c src/storage.c)
target_include_directories(task_core PUBLIC include)
add_executable(c-task-lab src/main.c src/input.c)
target_link_libraries(c-task-lab PRIVATE task_core)
include(CTest)
if(BUILD_TESTING)
    add_executable(test_core tests/test_core.c)
    target_link_libraries(test_core PRIVATE task_core)
    add_test(NAME core COMMAND test_core)
    add_executable(test_input tests/test_input.c src/input.c)
    target_include_directories(test_input PRIVATE include)
    add_test(NAME input COMMAND test_input)
endif()
# CTest 运行核心与输入测试；TODO 练习初始状态会失败，故不加入。
file(GLOB LESSON_SOURCES CONFIGURE_DEPENDS lessons/*.c)
foreach(source IN LISTS LESSON_SOURCES)
    get_filename_component(base ${source} NAME_WE)
    add_executable(lesson_${base} ${source})
endforeach()
